DRY RUN

- entire pipeline
python -m src.cli pipeline --config src/configs/pipeline.yaml --dry-run


FULL RUN

- single stage
# 1) fetch
python -m src.pipeline.fetch --config src/configs/fetch.yaml

# 2) merge (uses dataset or the fetch pointer if you set it)
python -m src.pipeline.merge --config src/configs/merge.yaml

# 3) validate (pre)
python -m src.pipeline.validate --config src/configs/validate_pre.yaml

# 4) cleanup (acts on latest pre report)
python -m src.pipeline.cleanup --config src/configs/cleanup.yaml

# 5) resize
python -m src.pipeline.resize --config src/configs/resize.yaml

# 6) validate (post)
python -m src.pipeline.validate --config src/configs/validate_post.yaml

# 7) split
python -m src.pipeline.split --config src/configs/split.yaml

# 8) train
python -m src.pipeline.train --config src/configs/train.yaml

# 9) evaluate
python -m src.pipeline.evaluate --config src/configs/evaluate.yaml


FULL END-TO-END

- entire pipline with all functions
python -m src.cli pipeline --config src/configs/pipeline.yaml

- This makes the fetch dry-run but run the full pipeline
python -m src.cli pipeline --config src/configs/pipeline.yaml -o fetch.dry_run=true

SKIP EARLY STAGE 
python -m src.cli pipeline --config src/configs/pipeline.yaml --skip fetch --skip merge


RESUME RESIZE
python -m src.cli pipeline --config src/configs/pipeline.yaml --resume-from resize


OVERRIDES VALUES ON THE FLY
python -m src.cli pipeline --config src/configs/pipeline.yaml \
  -o train.loop.epochs=5 \
  -o evaluate.io.make_gradcam=false
